# 16장 RNN과 어텐션을 사용한 자연어 처리

RNN(Recurrent Neural Network): 시계열 또는 순차 데이터를 예측하는 딥러닝을 위한 신경망 아키텍처

## 16.1 Char-RNN으로 셰익스피어 문체 생성하기

셰익스피어 희곡 텍스트 파일을 불러와 문자 단위로 처리

모델이 문장에서 다음 글자를 예측하도록 훈련

단어, 문법, 적절한 구두점 등을 학습

### 16.1.1 훈련 데이터셋 만들기

```python
import tensorflow as tf

# 셰익스피어 데이터 다운로드 및 불러오기
shakespeare_url = "https://homl.info/shakespeare"
filepath = tf.keras.utils.get_file("shakespeare.txt", shakespeare_url)
with open(filepath) as f:
    shakespeare_text = f.read()

# 텍스트 벡터화 (문자 단위로)
text_vec_layer = tf.keras.layers.TextVectorization(split="character", standardize="lower")
text_vec_layer.adapt([shakespeare_text])
encoded = text_vec_layer([shakespeare_text])[0]

# 토큰 개수 및 시퀀스 데이터셋 구성 함수 정의
n_tokens = text_vec_layer.vocabulary_size()
def to_dataset(sequence, length, shuffle=False, seed=None, batch_size=32):
    ds = tf.data.Dataset.from_tensor_slices(sequence)
    ds = ds.window(length + 1, shift=1, drop_remainder=True)
    ds = ds.flat_map(lambda window_ds: window_ds.batch(length + 1))
    if shuffle:
        ds = ds.shuffle(buffer_size=100_000, seed=seed)
    ds = ds.batch(batch_size)
    return ds.map(lambda window: (window[:, :-1], window[:, 1:])).prefetch(1)

# 시퀀스 길이 및 시드 설정
length = 100
tf.random.set_seed(42)

# 훈련, 검증, 테스트 세트 나누기
train_set = to_dataset(encoded[:1_000_000], length=length, shuffle=True, seed=42)
valid_set = to_dataset(encoded[1_000_000:1_060_000], length=length)
test_set = to_dataset(encoded[1_060_000:], length=length)
```

### 16.1.2 Char-RNN 모델 만들고 훈련하기

```python
# 모델 구성
model = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=n_tokens, output_dim=16),  # 임베딩 레이어
    tf.keras.layers.GRU(128, return_sequences=True),               # GRU 층
    tf.keras.layers.Dense(n_tokens, activation="softmax")         # 출력층 (다음 문자 확률 예측)
])

# 컴파일
model.compile(loss="sparse_categorical_crossentropy",
              optimizer="nadam",
              metrics=["accuracy"])

# 체크포인트 콜백
model_ckpt = tf.keras.callbacks.ModelCheckpoint(
    "my_shakespeare_model",
    monitor="val_accuracy",
    save_best_only=True
)

# 학습
history = model.fit(train_set,
                    validation_data=valid_set,
                    epochs=10,
                    callbacks=[model_ckpt])

# 전체 모델 구성 (벡터화 + 토큰 조정 + 학습된 모델)
shakespeare_model = tf.keras.Sequential([
    text_vec_layer,                          # 문자 벡터화 레이어
    tf.keras.layers.Lambda(lambda X: X - 2),  # PAD, OOV 토큰 제거
    model                                     # 앞에서 만든 모델
])

# 예측 실행
y_proba = shakespeare_model.predict(["To be or not to b"])[0, -1]
y_pred = tf.argmax(y_proba)
text_vec_layer.get_vocabulary()[y_pred + 2]  # 예측한 다음 문자
```

### 16.1.3 가짜 셰익스피어 텍스트 생성하기

그리디 디코딩: 먼저 초기 텍스트를 주입하고 모델이 가장 가능성 있는 다음 글자를 예측, 그런 다음 예측한 글자를 텍스트 끝에 추가하고 늘어난 텍스트를 모델에 전달하여 다음 글자를 예측

뉴클리어스 샘플링: 더 설득력 있는 텍스트를 생성하기 위한 일반적인 기법으로 상위 k개의 문자에서 샘플링하거나 총 확률이 특정 임곗값을 초과하는 가장 작은 상위 문자 집합에서만 샘플

```python
import tensorflow as tf
import numpy as np

# 확률 분포를 로그 스케일로 변환
log_probas = tf.math.log([[0.5, 0.4, 0.1]])
tf.random.set_seed(42)
# 확률 분포에 따라 샘플링
tf.random.categorical(log_probas, num_samples=8)
# 출력 결과: <tf.Tensor: shape=(1, 8), dtype=int64, numpy=array([[0, 2, 0, 0, 0, 0, 0, 1]])>

# 다음 글자를 예측하는 함수
def next_char(text, temperature=1):
    y_proba = shakespeare_model.predict([text])[0, -1]
    rescaled_logits = tf.math.log(y_proba) / temperature
    char_id = tf.random.categorical(rescaled_logits[tf.newaxis], num_samples=1)[0, 0]
    return text_vec_layer.get_vocabulary()[char_id + 2]

# 여러 글자 생성하는 함수
def extend_text(text, n_chars=50, temperature=1):
    for _ in range(n_chars):
        text += next_char(text, temperature)
    return text

# 샘플 텍스트 생성 테스트
tf.random.set_seed(42)
print(extend_text("To be or not to be", temperature=0.01))
print(extend_text("To be or not to be", temperature=1))
print(extend_text("To be or not to be", temperature=100))
```

### 16.1.4 상태가 있는 RNN

상태가 없는 RNN은 훈련 반복마다 모델의 은닉 상태를 0으로 초기화

타임 스텝마다 이 상태를 업데이트하고 마지막 타임 스텝 후에는 더는 필요가 없기 때문에 버림

상태가 있는 RNN은 한 훈련 배치를 처리한 후에 마지막 상태를 다음 훈련 배치의 초기 상태로 사용

이러면 역전파는 짧은 시퀀스에서 일어나지만 모델이 장기간 패턴을 학습 가능

```python
# 데이터셋 생성 함수
def to_dataset_for_stateful_rnn(sequence, length):
    ds = tf.data.Dataset.from_tensor_slices(sequence)
    ds = ds.window(length + 1, shift=length, drop_remainder=True)
    ds = ds.flat_map(lambda window: window.batch(length + 1)).batch(1)
    return ds.map(lambda window: (window[:, :-1], window[:, 1:])).prefetch(1)

# 학습, 검증, 테스트용 데이터셋 생성
stateful_train_set = to_dataset_for_stateful_rnn(encoded[:1_000_000], length)
stateful_valid_set = to_dataset_for_stateful_rnn(encoded[1_000_000:1_060_000], length)
stateful_test_set = to_dataset_for_stateful_rnn(encoded[1_060_000:], length)

# stateful RNN 모델 정의
model = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=n_tokens, output_dim=16,
                              batch_input_shape=[1, None]),
    tf.keras.layers.GRU(128, return_sequences=True, stateful=True),
    tf.keras.layers.Dense(n_tokens, activation="softmax")
])

# stateful RNN을 위한 Callback 클래스 정의
class ResetStatesCallback(tf.keras.callbacks.Callback):
    def on_epoch_begin(self, epoch, logs):
        self.model.reset_states()

# 모델 컴파일 및 훈련
model.compile(loss="sparse_categorical_crossentropy", optimizer="nadam",
              metrics=["accuracy"])
history = model.fit(stateful_train_set,
                    validation_data=stateful_valid_set,
                    epochs=10,
                    callbacks=[ResetStatesCallback(), model_ckpt])
```

### 16.2.1 마스킹

문장의 길이가 다를 경우, 모델이 패딩(padding) 된 부분을 무시하도록 처리

```python
# 입력: 문자열을 받는 입력 레이어
inputs = tf.keras.layers.Input(shape=(), dtype=tf.string)

# 텍스트 벡터화
token_ids = text_vec_layer(inputs)

# 마스크 생성 (padding == 0 위치 마스킹)
mask = tf.math.not_equal(token_ids, 0)

# 임베딩 후 GRU 레이어 적용
Z = tf.keras.layers.Embedding(vocab_size, embed_size)(token_ids)
Z = tf.keras.layers.GRU(128, dropout=0.2)(Z, mask=mask)

# 출력층: 이진 분류 (시그모이드)
outputs = tf.keras.layers.Dense(1, activation="sigmoid")(Z)

# 모델 정의
model = tf.keras.Model(inputs=inputs, outputs=outputs)

# TextVectorization을 ragged 텐서로 사용 (패딩 대신 가변 길이 표현)
text_vec_layer_ragged = tf.keras.layers.TextVectorization(
    max_tokens=vocab_size,
    ragged=True
)

# 데이터셋을 기반으로 어휘집 구축
text_vec_layer_ragged.adapt(train_set.map(lambda reviews, labels: reviews))

# ragged 텐서 출력 예시
text_vec_layer_ragged(["Great movie!", "This is DiCaprio's best role."])
# 결과: <tf.RaggedTensor [[186, 18], [11, 7, 1, 116, 217]]>
```

## 16.3 신경망 기계 번역을 위한 인코더-디코더 네트워크

티처 포싱: 실제 무엇이 출력되는지 상상하지 않고 디코더는 이전 스텝에서 출력되어야 할 단어를 입력으로 사용하여 모델의 성능을 향상하고 훈련 속도를 크게 높이는 기법

![image](https://github.com/user-attachments/assets/935caee0-ed2d-44e9-97b1-a2619e9168fa)

![image](https://github.com/user-attachments/assets/bcb16f55-1139-4722-9e86-0e9e4c4db4c9)

```python
import os
import numpy as np
import tensorflow as tf
from pathlib import Path

# 1. 데이터 다운로드
url = "https://storage.googleapis.com/download.tensorflow.org/data/spa-eng.zip"
path = tf.keras.utils.get_file("spa-eng.zip", origin=url, cache_dir="datasets", extract=True)
text = (Path(path).with_name("spa-eng") / "spa.txt").read_text()

# 2. 전처리 및 문장쌍 추출
text = text.replace("¡", "").replace("¿", "")
pairs = [line.split("\t") for line in text.splitlines()]
np.random.shuffle(pairs)
sentences_en, sentences_es = zip(*pairs)

# 3. 데이터 확인
for i in range(3):
    print(sentences_en[i], "→", sentences_es[i])

# 4. TextVectorization 설정
vocab_size = 10000
max_length = 50

text_vec_layer_en = tf.keras.layers.TextVectorization(
    vocab_size=vocab_size,
    output_sequence_length=max_length
)
text_vec_layer_es = tf.keras.layers.TextVectorization(
    vocab_size=vocab_size,
    output_sequence_length=max_length
)

# 5. 어휘집 구축
text_vec_layer_en.adapt(sentences_en)
text_vec_layer_es.adapt(["startofseq " + s + " endofseq" for s in sentences_es])

# 6. 학습/검증 세트 분할 및 벡터화
X_train = tf.constant(sentences_en[:100_000])
X_valid = tf.constant(sentences_en[100_000:])

X_train_dec = tf.constant(["startofseq " + s for s in sentences_es[:100_000]])
X_valid_dec = tf.constant(["startofseq " + s for s in sentences_es[100_000:]])

Y_train = text_vec_layer_es([s + " endofseq" for s in sentences_es[:100_000]])
Y_valid = text_vec_layer_es([s + " endofseq" for s in sentences_es[100_000:]])

# 7. 인코더/디코더 입력
encoder_inputs = tf.keras.layers.Input(shape=(), dtype=tf.string)
decoder_inputs = tf.keras.layers.Input(shape=(), dtype=tf.string)

# 8. 임베딩 처리
embed_size = 128

encoder_input_ids = text_vec_layer_en(encoder_inputs)
decoder_input_ids = text_vec_layer_es(decoder_inputs)

encoder_embedding_layer = tf.keras.layers.Embedding(vocab_size, embed_size, mask_zero=True)
decoder_embedding_layer = tf.keras.layers.Embedding(vocab_size, embed_size, mask_zero=True)

encoder_embeddings = encoder_embedding_layer(encoder_input_ids)
decoder_embeddings = decoder_embedding_layer(decoder_input_ids)

# 9. 인코더 LSTM
encoder = tf.keras.layers.LSTM(512, return_state=True)
encoder_outputs, state_h, state_c = encoder(encoder_embeddings)
encoder_state = [state_h, state_c]

# 10. 디코더 LSTM
decoder = tf.keras.layers.LSTM(512, return_sequences=True)
decoder_outputs = decoder(decoder_embeddings, initial_state=encoder_state)

# 11. 출력층
output_layer = tf.keras.layers.Dense(vocab_size, activation="softmax")
Y_proba = output_layer(decoder_outputs)

# 12. 모델 컴파일 및 학습
model = tf.keras.Model(inputs=[encoder_inputs, decoder_inputs], outputs=[Y_proba])
model.compile(loss="sparse_categorical_crossentropy", optimizer="nadam", metrics=["accuracy"])

model.fit(
    (X_train, X_train_dec), Y_train,
    validation_data=((X_valid, X_valid_dec), Y_valid),
    epochs=10
)

# 13. 번역 함수 정의
def translate(sentence_en):
    translation = ""
    for word_idx in range(max_length):
        x = np.array([sentence_en])
        x_dec = np.array(["startofseq " + translation])
        y_proba = model.predict((x, x_dec))[0, word_idx]
        predicted_word_id = np.argmax(y_proba)
        predicted_word = text_vec_layer_es.get_vocabulary()[predicted_word_id]
        if predicted_word == "endofseq":
            break
        translation += " " + predicted_word
    return translation.strip()
```

#### 출력 층 최적화

샘플링 소프트맥스: 모델이 타깃 단어에 대한 로짓과 타깃이 아닌 단어 중 랜덤하게 샘플링한 단어의 로짓을 살펴보고, 이 로짓만을 기반으로 손실의 근사치 계산산

### 16.3.1 양방향 RNN

각 타임 스텝에서 일반적인 순환 층은 과거와 현재의 입력만 보고 출력을 생성하는 인과적 관계임

-> 미래를 볼 수 없음

하나는 왼쪽에서 오른쪽으로 단어를 읽고 다른 하나는 오른쪽에서 왼쪽으로 읽고

그다음 일반적으로 타임 스텝마다 이 두 출력을 연결

```python
# 양방향 LSTM 인코더 정의
encoder = tf.keras.layers.Bidirectional(
    tf.keras.layers.LSTM(256, return_state=True)
)

# 인코더 실행: 출력과 상태 벡터 추출
encoder_outputs, *encoder_state = encoder(encoder_embeddings)

# 양방향 상태 벡터 연결 (forward & backward)
encoder_state = [
    tf.concat(encoder_state[::2], axis=-1),  # 은닉 상태 (0번 & 2번)
    tf.concat(encoder_state[1::2], axis=-1)  # 셀 상태 (1번 & 3번)
]
```

## 16.4 어텐션 메커니즘

번역이 이어지는 경로가 매우 김(실제 사용되기 전에 여러 단계를 이동)

각 타임 스텝에서 적절한 단어에 디코더가 초점을 맞춤

-> 경로가 훨씬 짧아져 RNN의 단기 기억 제한성이 훨씬 적은 영향을 미침

![image](https://github.com/user-attachments/assets/75dce570-9059-41b4-adca-1038613f2db8)

```python
# 양방향 LSTM 인코더 (시퀀스 출력 + 상태값 반환)
encoder = tf.keras.layers.Bidirectional(
    tf.keras.layers.LSTM(256, return_sequences=True, return_state=True)
)

# 어텐션 레이어 적용
attention_layer = tf.keras.layers.Attention()
attention_outputs = attention_layer([decoder_outputs, encoder_outputs])

# 최종 출력층
output_layer = tf.keras.layers.Dense(vocab_size, activation="softmax")
Y_proba = output_layer(attention_outputs)

# 예시 번역 실행
translate("I like soccer and also going to the beach")
# → 'me gusta el futbol y tambien ir a la playa'
```

### 16.4.1 트렌스포머 구조: 어텐션만 있으면 된다

순환 층이나 합성곱 층을 전혀 사용하지 않고 어텐션 메커니즘만 사용하여 NMT 분야의 최신 기술을 크게 향상

순환하지 않기 떄문에 RNN처럼 그레이디언트 소실이나 폭주 문제가 발생하지 않으며, 더 적은 단계로 학습할 수 있고, 여러 GPU에 걸쳐 병렬화하기 쉬우며, RNN보다 장기 패턴을 더 잘 포착할 수 있음

![image](https://github.com/user-attachments/assets/d766f818-19f5-44d8-b50f-58794eafb61e)

#### 위치 인코딩

문장 안에 있는 단어의 위치를 인코딩한 밀집 벡터 

단순히 i번째 위치 인코딩이 문장에 있는 i번째 단어의 단어 임베딩에 더해짐

```python
max_length = 50  # 전체 문장의 최대 길이
embed_size = 128

# 위치 임베딩 레이어 생성
pos_embed_layer = tf.keras.layers.Embedding(max_length, embed_size)

# 인코더 위치 인코딩 추가
batch_max_len_enc = tf.shape(encoder_embeddings)[1]
encoder_in = encoder_embeddings + pos_embed_layer(tf.range(batch_max_len_enc))

# 디코더 위치 인코딩 추가
batch_max_len_dec = tf.shape(decoder_embeddings)[1]
decoder_in = decoder_embeddings + pos_embed_layer(tf.range(batch_max_len_dec))

class PositionalEncoding(tf.keras.layers.Layer):
    def __init__(self, max_length, embed_size, dtype=tf.float32, **kwargs):
        super().__init__(dtype=dtype, **kwargs)
        assert embed_size % 2 == 0, "embed_size must be even"

        p, i = np.meshgrid(np.arange(max_length), 2 * np.arange(embed_size // 2))
        pos_emb = np.empty((1, max_length, embed_size))
        pos_emb[0, :, 0::2] = np.sin(p / 10000 ** (i / embed_size)).T
        pos_emb[0, :, 1::2] = np.cos(p / 10000 ** (i / embed_size)).T

        self.pos_encodings = tf.constant(pos_emb.astype(dtype))
        self.supports_masking = True

    def call(self, inputs):
        batch_max_length = tf.shape(inputs)[1]
        return inputs + self.pos_encodings[:, :batch_max_length]

# 사용 예시
pos_embed_layer = PositionalEncoding(max_length, embed_size)
encoder_in = pos_embed_layer(encoder_embeddings)
decoder_in = pos_embed_layer(decoder_embeddings)
```

#### 멀티 헤드 어텐션

기본: 스케일드 점곱 어텐션

멀티 헤드 어텐션: 스케일드 점곱 어텐션 층의 묶음

각 층은 값, 키, 쿼리에 대한 선형 변환을 먼저 수행

출력은 단순히 모두 연결되어 마지막 선형 변환을 통
![image](https://github.com/user-attachments/assets/457113c0-9fe0-48a7-9c14-eb05cb26b148)

```python
N = 2  # 인코더 블록 수
num_heads = 8
dropout_rate = 0.1
n_units = 128

# 패딩 마스크
encoder_pad_mask = tf.math.not_equal(encoder_input_ids, 0)[:, tf.newaxis]

Z = encoder_in
for _ in range(N):
    skip = Z

    # 멀티헤드 어텐션
    attn_layer = tf.keras.layers.MultiHeadAttention(
        num_heads=num_heads,
        key_dim=embed_size,
        dropout=dropout_rate
    )
    Z = attn_layer(Z, Z, attention_mask=encoder_pad_mask)
    Z = tf.keras.layers.LayerNormalization()(tf.keras.layers.Add()([Z, skip]))

    # 피드포워드 네트워크
    skip = Z
    Z = tf.keras.layers.Dense(n_units, activation="relu")(Z)
    Z = tf.keras.layers.Dense(embed_size)(Z)
    Z = tf.keras.layers.Dropout(dropout_rate)(Z)
    Z = tf.keras.layers.LayerNormalization()(tf.keras.layers.Add()([Z, skip]))

# 패딩 및 미래 마스크
decoder_pad_mask = tf.math.not_equal(decoder_input_ids, 0)[:, tf.newaxis]
causal_mask = tf.linalg.band_part(
    tf.ones((batch_max_len_dec, batch_max_len_dec), tf.bool), -1, 0
)

Z = decoder_in
for _ in range(N):
    skip = Z

    # self-attention (causal mask 적용)
    attn_layer = tf.keras.layers.MultiHeadAttention(
        num_heads=num_heads,
        key_dim=embed_size,
        dropout=dropout_rate
    )
    Z = attn_layer(Z, Z, attention_mask=causal_mask & decoder_pad_mask)
    Z = tf.keras.layers.LayerNormalization()(tf.keras.layers.Add()([Z, skip]))

    skip = Z

    # encoder-decoder attention
    attn_layer = tf.keras.layers.MultiHeadAttention(
        num_heads=num_heads,
        key_dim=embed_size,
        dropout=dropout_rate
    )
    Z = attn_layer(Z, encoder_outputs, attention_mask=decoder_pad_mask)
    Z = tf.keras.layers.LayerNormalization()(tf.keras.layers.Add()([Z, skip]))

    # 피드포워드
    skip = Z
    Z = tf.keras.layers.Dense(n_units, activation="relu")(Z)
    Z = tf.keras.layers.Dense(embed_size)(Z)
    Z = tf.keras.layers.Dropout(dropout_rate)(Z)
    Z = tf.keras.layers.LayerNormalization()(tf.keras.layers.Add()([Z, skip]))

# 출력층
Y_proba = tf.keras.layers.Dense(vocab_size, activation="softmax")(Z)

# 모델 정의 및 학습
model = tf.keras.Model(inputs=[encoder_inputs, decoder_inputs], outputs=[Y_proba])
model.compile(loss="sparse_categorical_crossentropy", optimizer="nadam", metrics=["accuracy"])
model.fit(
    (X_train, X_train_dec), Y_train,
    validation_data=((X_valid, X_valid_dec), Y_valid),
    epochs=10
)
```

## 16.5 언어 모델 분야의 최근 혁신

BERT 모델의 강점을 잘 설명하는 두 가지 사전 훈련 작업

1. 마스크드 언어 모델(MLM): 문장에 있는 각 단어는 15% 확률로 마스킹. 모델은 마스킹된 단어를 예측하도록 훈련

2. 다음 문장 예측(NSP): 두 문장이 연속적인지 아닌지를 예측하도록 모델 훈련

![image](https://github.com/user-attachments/assets/3181ac0d-ab85-40e5-9fcd-f83df90db2d2)

제로 샷 학습(ZSL):  테스트 시 학습자가 훈련 중에 관찰되지 않은 클래스의 샘플을 관찰하고 해당 샘플이 속한 클래스를 예측

## 16.6 비전 트렌스포머(ViT)

이미지를 16*16의 작은 정사각형으로 자르고 정사각형의 시퀀스를 마치 단어 표현의 시퀀스처럼 취급

더 정확히 말하면 먼저 정사각형을 16*16*3차원 벡터로 평평하게 만든 다음

벡터의 차원을 유지하면서 변환하는 선형 층에 통과시킴

이로부터 얻은 벡터 시퀀스를 단어 임베딩 시퀀스처럼 처리 가능

->

데이터 효율적인 이미지 트랜스포머(Deit): ViT에서 발전된 모델로 추가 훈련 데이터 없이도 사용 가능

->

멀티모달 트랜스포머: 텍스트, 이미지, 오디오 등 거의 모든 종류의 데이터 입력 가능

퍼시비어는 비교적 상당히 짧은 입력의 잠재 표현을 점진적으로 개선

->

DINO: 자기 지도 방식으로 레이블 없이 훈련되고 높은 정확도의 시맨틱 분할이 가능한 비전 트랜스포머

훈련 중에 복제되어 한 네트워크는 티처 역할을 하고 다른 네트워크는 스튜던트 역살을 함

->

데이터의 양에 따라 ViT를 확장하거나 축소

-> 

여러 개의 트랜스포머를 먼저 학습시킨 다음 가중치를 평균하여 새롭고 개선된 모델 생성

->

최신 트렌드: 제로 샷 또는 퓨 샷 학습이 가능한 대규모 멀티모달 모델을 만드는 것

#### 설명 가능성

어텐션 메커니즘의 추가적인 장점으로 모델이 출력을 만들게 된 원인이 무엇인지 이해하기 쉬움

이는 모델이 올바르게 예측하지 못할 때 특히 유용

## 16.7 허깅 페이스와 트랜스포머스 라이브러리

허깅 페이스: 자연어 처리, 비전 등을 위해 사용하기 쉬운 오픈 소스 도구 생태계 구축한 AI 회사

```python
from transformers import pipeline

# 1. 감정 분석 파이프라인 생성
classifier = pipeline("sentiment-analysis")

# 2. 문장 하나 분석
result = classifier("The actors were very convincing.")

# 3. 여러 문장 한 번에 분석
classifier(["I am from India.", "I am from Iraq."])

from transformers import pipeline

# 4. MNLI 모델을 이용한 텍스트 관계 분류 (모순인지 확인)
model_name = "huggingface/distilbert-base-uncased-finetuned-mnli"
classifier_mnli = pipeline("text-classification", model=model_name)

# 5. 문장 쌍 간의 관계 분석 (entailment, contradiction, neutral)
classifier_mnli("She loves me. [SEP] She loves me not.")

from transformers import AutoTokenizer, TFAutoModelForSequenceClassification

# 6. MNLI 모델과 토크나이저 불러오기
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = TFAutoModelForSequenceClassification.from_pretrained(model_name)

# 7. 문장 쌍을 [SEP]로 구분하여 입력 토큰화
token_ids = tokenizer(
    ["I like soccer. [SEP] We all love soccer.",
     "Joe lived for a very long time. [SEP] Joe is old."],
    padding=True,
    return_tensors="tf"
)

# 8. 모델에 입력하여 로짓값 출력
outputs = model(token_ids)

import tensorflow as tf

# 9. 소프트맥스로 클래스별 확률 계산
Y_probas = tf.keras.activations.softmax(outputs.logits)

# 10. 가장 높은 확률의 클래스를 예측 결과로 선택
Y_pred = tf.argmax(Y_probas, axis=1)  # 0 = entailment, 1 = neutral, 2 = contradiction

# 11. 문장 쌍을 분류 문제로 학습하는 예시
sentences = [("Sky is blue", "Sky is red"), ("I love her", "She loves me")]
X_train = tokenizer(sentences, padding=True, return_tensors="tf").data
Y_train = tf.constant([2, 1])  # 2 = contradiction, 1 = neutral

# 12. 분류 손실 함수와 모델 컴파일
loss = tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True)
model.compile(loss=loss, optimizer="adam", metrics=["accuracy"])

# 13. 모델 학습 수행
history = model.fit(X_train, Y_train, epochs=2)
```

#### 편견과 공정성

바람직하지 않은 편향은 일반적으로 훈련 데이터 자체에서 비롯됨

미세 튜닝 시 중립 클래스를 추가하면 편향이 대부분 사라지지만 훈련 데이터가 편향의 유일한 원인은 아님

모델의 구조, 훈련에 사용된 손실 또는 정규화 종류, 최적화 도구 등 모든 것이 모델이 최종적으로 학습하는 내용에 영향을 미침
