# 16장 RNN과 어텐션을 사용한 자연어 처리

RNN(Recurrent Neural Network): 시계열 또는 순차 데이터를 예측하는 딥러닝을 위한 신경망 아키텍처

## 16.1 Char-RNN으로 셰익스피어 문체 생성하기

셰익스피어 희곡 텍스트 파일을 불러와 문자 단위로 처리

모델이 문장에서 다음 글자를 예측하도록 훈련

단어, 문법, 적절한 구두점 등을 학습

### 16.1.1 훈련 데이터셋 만들기

```python
import tensorflow as tf

# 셰익스피어 데이터 다운로드 및 불러오기
shakespeare_url = "https://homl.info/shakespeare"
filepath = tf.keras.utils.get_file("shakespeare.txt", shakespeare_url)
with open(filepath) as f:
    shakespeare_text = f.read()

# 텍스트 벡터화 (문자 단위로)
text_vec_layer = tf.keras.layers.TextVectorization(split="character", standardize="lower")
text_vec_layer.adapt([shakespeare_text])
encoded = text_vec_layer([shakespeare_text])[0]

# 토큰 개수 및 시퀀스 데이터셋 구성 함수 정의
n_tokens = text_vec_layer.vocabulary_size()
def to_dataset(sequence, length, shuffle=False, seed=None, batch_size=32):
    ds = tf.data.Dataset.from_tensor_slices(sequence)
    ds = ds.window(length + 1, shift=1, drop_remainder=True)
    ds = ds.flat_map(lambda window_ds: window_ds.batch(length + 1))
    if shuffle:
        ds = ds.shuffle(buffer_size=100_000, seed=seed)
    ds = ds.batch(batch_size)
    return ds.map(lambda window: (window[:, :-1], window[:, 1:])).prefetch(1)

# 시퀀스 길이 및 시드 설정
length = 100
tf.random.set_seed(42)

# 훈련, 검증, 테스트 세트 나누기
train_set = to_dataset(encoded[:1_000_000], length=length, shuffle=True, seed=42)
valid_set = to_dataset(encoded[1_000_000:1_060_000], length=length)
test_set = to_dataset(encoded[1_060_000:], length=length)
```

### 16.1.2 Char-RNN 모델 만들고 훈련하기

```python
# 모델 구성
model = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=n_tokens, output_dim=16),  # 임베딩 레이어
    tf.keras.layers.GRU(128, return_sequences=True),               # GRU 층
    tf.keras.layers.Dense(n_tokens, activation="softmax")         # 출력층 (다음 문자 확률 예측)
])

# 컴파일
model.compile(loss="sparse_categorical_crossentropy",
              optimizer="nadam",
              metrics=["accuracy"])

# 체크포인트 콜백
model_ckpt = tf.keras.callbacks.ModelCheckpoint(
    "my_shakespeare_model",
    monitor="val_accuracy",
    save_best_only=True
)

# 학습
history = model.fit(train_set,
                    validation_data=valid_set,
                    epochs=10,
                    callbacks=[model_ckpt])

# 전체 모델 구성 (벡터화 + 토큰 조정 + 학습된 모델)
shakespeare_model = tf.keras.Sequential([
    text_vec_layer,                          # 문자 벡터화 레이어
    tf.keras.layers.Lambda(lambda X: X - 2),  # PAD, OOV 토큰 제거
    model                                     # 앞에서 만든 모델
])

# 예측 실행
y_proba = shakespeare_model.predict(["To be or not to b"])[0, -1]
y_pred = tf.argmax(y_proba)
text_vec_layer.get_vocabulary()[y_pred + 2]  # 예측한 다음 문자
```

### 16.1.3 가짜 셰익스피어 텍스트 생성하기

그리디 디코딩: 먼저 초기 텍스트를 주입하고 모델이 가장 가능성 있는 다음 글자를 예측, 그런 다음 예측한 글자를 텍스트 끝에 추가하고 늘어난 텍스트를 모델에 전달하여 다음 글자를 예측

뉴클리어스 샘플링: 더 설득력 있는 텍스트를 생성하기 위한 일반적인 기법으로 상위 k개의 문자에서 샘플링하거나 총 확률이 특정 임곗값을 초과하는 가장 작은 상위 문자 집합에서만 샘플

```python
import tensorflow as tf
import numpy as np

# 확률 분포를 로그 스케일로 변환
log_probas = tf.math.log([[0.5, 0.4, 0.1]])
tf.random.set_seed(42)
# 확률 분포에 따라 샘플링
tf.random.categorical(log_probas, num_samples=8)
# 출력 결과: <tf.Tensor: shape=(1, 8), dtype=int64, numpy=array([[0, 2, 0, 0, 0, 0, 0, 1]])>

# 다음 글자를 예측하는 함수
def next_char(text, temperature=1):
    y_proba = shakespeare_model.predict([text])[0, -1]
    rescaled_logits = tf.math.log(y_proba) / temperature
    char_id = tf.random.categorical(rescaled_logits[tf.newaxis], num_samples=1)[0, 0]
    return text_vec_layer.get_vocabulary()[char_id + 2]

# 여러 글자 생성하는 함수
def extend_text(text, n_chars=50, temperature=1):
    for _ in range(n_chars):
        text += next_char(text, temperature)
    return text

# 샘플 텍스트 생성 테스트
tf.random.set_seed(42)
print(extend_text("To be or not to be", temperature=0.01))
print(extend_text("To be or not to be", temperature=1))
print(extend_text("To be or not to be", temperature=100))
```

### 16.1.4 상태가 있는 RNN

상태가 없는 RNN은 훈련 반복마다 모델의 은닉 상태를 0으로 초기화

타임 스텝마다 이 상태를 업데이트하고 마지막 타임 스텝 후에는 더는 필요가 없기 때문에 버림

상태가 있는 RNN은 한 훈련 배치를 처리한 후에 마지막 상태를 다음 훈련 배치의 초기 상태로 사용

이러면 역전파는 짧은 시퀀스에서 일어나지만 모델이 장기간 패턴을 학습 가능

```python
# 데이터셋 생성 함수
def to_dataset_for_stateful_rnn(sequence, length):
    ds = tf.data.Dataset.from_tensor_slices(sequence)
    ds = ds.window(length + 1, shift=length, drop_remainder=True)
    ds = ds.flat_map(lambda window: window.batch(length + 1)).batch(1)
    return ds.map(lambda window: (window[:, :-1], window[:, 1:])).prefetch(1)

# 학습, 검증, 테스트용 데이터셋 생성
stateful_train_set = to_dataset_for_stateful_rnn(encoded[:1_000_000], length)
stateful_valid_set = to_dataset_for_stateful_rnn(encoded[1_000_000:1_060_000], length)
stateful_test_set = to_dataset_for_stateful_rnn(encoded[1_060_000:], length)

# stateful RNN 모델 정의
model = tf.keras.Sequential([
    tf.keras.layers.Embedding(input_dim=n_tokens, output_dim=16,
                              batch_input_shape=[1, None]),
    tf.keras.layers.GRU(128, return_sequences=True, stateful=True),
    tf.keras.layers.Dense(n_tokens, activation="softmax")
])

# stateful RNN을 위한 Callback 클래스 정의
class ResetStatesCallback(tf.keras.callbacks.Callback):
    def on_epoch_begin(self, epoch, logs):
        self.model.reset_states()

# 모델 컴파일 및 훈련
model.compile(loss="sparse_categorical_crossentropy", optimizer="nadam",
              metrics=["accuracy"])
history = model.fit(stateful_train_set,
                    validation_data=stateful_valid_set,
                    epochs=10,
                    callbacks=[ResetStatesCallback(), model_ckpt])
```

부분 단어 규제: 훈련 중에 토큰화에 약간의 무작위성을 도입하여 정확도와 견고성 향상

```python
raw_train_set, raw_valid_set, raw_test_set = tfds.load(
    name="imdb_reviews",
    split=["train[:90%]", "train[90%:]", "test"],
    as_supervised=True
)

train_set = raw_train_set.shuffle(5000, seed=42).batch(32).prefetch(1)
valid_set = raw_valid_set.batch(32).prefetch(1)
test_set = raw_test_set.batch(32).prefetch(1)

vocab_size = 1000
text_vec_layer = tf.keras.layers.TextVectorization(max_tokens=vocab_size)
text_vec_layer.adapt(train_set.map(lambda reviews, labels: reviews))

model = tf.keras.Sequential([
    text_vec_layer,
    tf.keras.layers.Embedding(vocab_size, embed_size),
    tf.keras.layers.GRU(128),
    tf.keras.layers.Dense(1, activation="sigmoid")
])

model.compile(loss="binary_crossentropy", optimizer="nadam", metrics=["accuracy"])
history = model.fit(train_set, validation_data=valid_set, epochs=2)
```

## 16.5 언어 모델 분야의 최근 혁신

BERT 모델의 강점을 잘 설명하는 두 가지 사전 훈련 작업

1. 마스크드 언어 모델(MLM): 문장에 있는 각 단어는 15% 확률로 마스킹. 모델은 마스킹된 단어를 예측하도록 훈련

2. 다음 문장 예측(NSP): 두 문장이 연속적인지 아닌지를 예측하도록 모델 훈련

![image](https://github.com/user-attachments/assets/3181ac0d-ab85-40e5-9fcd-f83df90db2d2)

제로 샷 학습(ZSL):  테스트 시 학습자가 훈련 중에 관찰되지 않은 클래스의 샘플을 관찰하고 해당 샘플이 속한 클래스를 예측

## 16.6 비전 트렌스포머(ViT)

이미지를 16*16의 작은 정사각형으로 자르고 정사각형의 시퀀스를 마치 단어 표현의 시퀀스처럼 취급

더 정확히 말하면 먼저 정사각형을 16*16*3차원 벡터로 평평하게 만든 다음

벡터의 차원을 유지하면서 변환하는 선형 층에 통과시킴

이로부터 얻은 벡터 시퀀스를 단어 임베딩 시퀀스처럼 처리 가능

->

데이터 효율적인 이미지 트랜스포머(Deit): ViT에서 발전된 모델로 추가 훈련 데이터 없이도 사용 가능

->

멀티모달 트랜스포머: 텍스트, 이미지, 오디오 등 거의 모든 종류의 데이터 입력 가능

퍼시비어는 비교적 상당히 짧은 입력의 잠재 표현을 점진적으로 개선

->

DINO: 자기 지도 방식으로 레이블 없이 훈련되고 높은 정확도의 시맨틱 분할이 가능한 비전 트랜스포머

훈련 중에 복제되어 한 네트워크는 티처 역할을 하고 다른 네트워크는 스튜던트 역살을 함

->

데이터의 양에 따라 ViT를 확장하거나 축소

-> 

여러 개의 트랜스포머를 먼저 학습시킨 다음 가중치를 평균하여 새롭고 개선된 모델 생성

->

최신 트렌드: 제로 샷 또는 퓨 샷 학습이 가능한 대규모 멀티모달 모델을 만드는 것

#### 설명 가능성

## 16.7 허깅 페이스와 트랜스포머스 라이브러리

허깅 페이스: 자연어 처리, 비전 등을 위해 사용하기 쉬운 오픈 소스 도구 생태계 구축한 AI 회사

```python
from transformers import pipeline

# 1. 감정 분석 파이프라인 생성
classifier = pipeline("sentiment-analysis")

# 2. 문장 하나 분석
result = classifier("The actors were very convincing.")

# 3. 여러 문장 한 번에 분석
classifier(["I am from India.", "I am from Iraq."])

from transformers import pipeline

# 4. MNLI 모델을 이용한 텍스트 관계 분류 (모순인지 확인)
model_name = "huggingface/distilbert-base-uncased-finetuned-mnli"
classifier_mnli = pipeline("text-classification", model=model_name)

# 5. 문장 쌍 간의 관계 분석 (entailment, contradiction, neutral)
classifier_mnli("She loves me. [SEP] She loves me not.")

from transformers import AutoTokenizer, TFAutoModelForSequenceClassification

# 6. MNLI 모델과 토크나이저 불러오기
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = TFAutoModelForSequenceClassification.from_pretrained(model_name)

# 7. 문장 쌍을 [SEP]로 구분하여 입력 토큰화
token_ids = tokenizer(
    ["I like soccer. [SEP] We all love soccer.",
     "Joe lived for a very long time. [SEP] Joe is old."],
    padding=True,
    return_tensors="tf"
)

# 8. 모델에 입력하여 로짓값 출력
outputs = model(token_ids)

import tensorflow as tf

# 9. 소프트맥스로 클래스별 확률 계산
Y_probas = tf.keras.activations.softmax(outputs.logits)

# 10. 가장 높은 확률의 클래스를 예측 결과로 선택
Y_pred = tf.argmax(Y_probas, axis=1)  # 0 = entailment, 1 = neutral, 2 = contradiction

# 11. 문장 쌍을 분류 문제로 학습하는 예시
sentences = [("Sky is blue", "Sky is red"), ("I love her", "She loves me")]
X_train = tokenizer(sentences, padding=True, return_tensors="tf").data
Y_train = tf.constant([2, 1])  # 2 = contradiction, 1 = neutral

# 12. 분류 손실 함수와 모델 컴파일
loss = tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True)
model.compile(loss=loss, optimizer="adam", metrics=["accuracy"])

# 13. 모델 학습 수행
history = model.fit(X_train, Y_train, epochs=2)
```

#### 편견과 공정성

바람직하지 않은 편향은 일반적으로 훈련 데이터 자체에서 비롯됨

미세 튜닝 시 중립 클래스를 추가하면 편향이 대부분 사라지지만 훈련 데이터가 편향의 유일한 원인은 아님

모델의 구조, 훈련에 사용된 손실 또는 정규화 종류, 최적화 도구 등 모든 것이 모델이 최종적으로 학습하는 내용에 영향을 미침
