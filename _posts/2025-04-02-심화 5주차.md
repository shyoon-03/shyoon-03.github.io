# 5장 사이파이를 사용한 기본 확률 및 통계 분석

## 5.1 사이파이로 데이터와 확률 간 관계 탐색하기

이항 분포(binomial distribution)

이항 검정(binomial test)

확률 질량 함수(Probability Mass Function)

시행 횟수가 증가할수록 분포도 중심이 오른쪽으로 이동하고 분포는 중심에서 더욱 분

```python
from scipy import stats

num_heads = 16
num_flips = 20
prob_head = 0.5

prob = stats.binom_test(num_heads, num_flips, prob_head)
print(f"15개 이상의 동전 앞면 또는 뒷면이 관측될 확률은 {prob:.17f}입니다.")

prob_16_heads = stats.binom.pmf(num_heads, num_flips, prob_head)
print(f"{num_heads}번의 앞면 중 {num_flips}번의 동전 뒤집기에서 관측될 확률은 {prob_16_heads}입니다.")

probabilities = stats.binom.pmf(range(21), num_flips, prob_head)
assert probabilities.tolist() == [prob_16_heads] * 2

interval_all_counts = range(21)
probabilities = stats.binom.pmf(interval_all_counts, num_flips, prob_head)
total_prob = probabilities.sum()
print(f"확률의 총합은 {total_prob:.14f}입니다.")

import matplotlib.pyplot as plt

plt.plot(interval_all_counts, probabilities)
plt.xlabel('동전 앞면의 동장 횟수')
plt.ylabel('확률')
plt.show()

flip_counts = [20, 80, 140, 200]
linestyles = ['-', '--', '-.', ':']
colors = ['b', 'g', 'r', 'k']

for num_flips, linestyle, color in zip(flip_counts, linestyles, colors):
    x_values = range(num_flips+1)
    y_values = stats.binom.pmf(x_values, num_flips, 0.5)
    plt.plot(x_values, y_values, linestyle=linestyle, color=color, label=f'{num_flips}회의 동전 뒤집기')

plt.legend()
plt.xlabel('동전 앞면의 동장 횟수')
plt.ylabel('확률')
plt.show()
```

## 5.2 중심성의 척도로서 평균

중앙값(median)

산술 평균(arithmetic mean)

```python
import numpy as np
measurements = np.array([80, 77, 73, 61, 74, 79, 81])

measurements.sort()
number_of_days = measurements.size
plt.plot(range(number_of_days), measurements)
plt.scatter(range(number_of_days), measurements)
plt.ylabel('온도')
plt.show()

difference = measurements.max() - measurements.min()
midpoint = measurements.min() + difference / 2
assert midpoint == (measurements.max() + measurements.min()) / 2
print(f"중간 온도는 {midpoint}도입니다.")

plt.plot(range(number_of_days), measurements)
plt.scatter(range(number_of_days), measurements)
plt.axhline(midpoint, color='k', linestyle='--')
plt.ylabel('온도')
plt.show()

median = measurements[3]
print(f"온도의 중앙값은 {median}도입니다.")
plt.plot(range(number_of_days), measurements)
plt.scatter(range(number_of_days), measurements)
plt.axhline(midpoint, color='k', linestyle='--', label='중간 점')
plt.axhline(median, color='g', linestyle='-.', label='중앙값')
plt.legend()
plt.ylabel('온도')
plt.show()

def squared_distance(value1, value2):
    return (value1 - value2) ** 2

possible_centers = range(measurements.min(), measurements.max() + 1)
penalties = [squared_distance(center, 61) for center in possible_centers]

plt.plot(possible_centers, penalties)
plt.scatter(possible_centers, penalties)
plt.xlabel('가능한 중심 값')
plt.ylabel('페널티')
plt.show()

def sum_of_squared_distances(value, measurements):
    return sum(squared_distance(value, m) for m in measurements)

penalties = [sum_of_squared_distances(center, measurements) for center in possible_centers]

plt.plot(possible_centers, penalties)
plt.scatter(possible_centers, penalties)
plt.xlabel('가능한 중심 값')
plt.ylabel('페널티')
plt.show()

least_penalized = 75
assert least_penalized == possible_centers[np.argmin(penalties)]

plt.plot(range(number_of_days), measurements)
plt.scatter(range(number_of_days), measurements)
plt.axhline(midpoint, color='k', linestyle='--', label='중간 점')
plt.axhline(median, color='g', linestyle='-.', label='중앙값')
plt.axhline(least_penalized, color='r', linestyle='-', label='페널티가 가장 낮은 중심')
plt.legend()
plt.ylabel('온도')
plt.show()

assert measurements.sum() / measurements.size == least_penalized

mean = measurements.mean()
assert mean == least_penalized
assert mean == np.mean(measurements)
assert mean == np.average(measurements)

equal_weights = [1] * 7
mean = np.average(measurements, weights=equal_weights)

unequal_weights = [100] + [1] * 6
assert mean != np.average(measurements, weights=unequal_weights)

weighted_mean = np.average([75, 77], weights=[9, 1])
print(f"평균은 {weighted_mean}입니다.")
assert weighted_mean == np.mean(9*[75] + [77])

assert weighted_mean == np.average([75, 77], weights=[900, 100])
assert weighted_mean == np.average([75, 77], weights=[0.9, 0.1])
```

## 5.3 흩어진 정도를 측정하는 분산

제곱합

분산

```python
california = np.array([52, 77, 96])
kentucky = np.array([71, 75, 79])
print(f"캘리포니아주의 평균 온도는 {california.mean()}입니다")
print(f"켄터키주의 평균 온도는 {kentucky.mean()}입니다")

plt.plot(range(3), california, color='b', label="캘리포니아")
plt.scatter(range(3), california, color='b')
plt.plot(range(3), kentucky, color='r', linestyle='--', label="켄터키")
plt.scatter(range(3), kentucky, color='r')
plt.axhline(75, color="k", linestyle='--', label="평균")
plt.legend()
plt.show()

def sum_of_squares(data):
    mean = np.mean(data)
    return sum(squared_distance(value, mean) for value in data)

california_sum_squares = sum_of_squares(california)
print(f"캘리포니아주의 온도의 제곱합은 {california_sum_squares}입니다")

kentucky_sum_squares = sum_of_squares(kentucky)
print(f"켄터키주의 온도의 제곱합은 {kentucky_sum_squares}입니다")

california_duplicated = np.array(california.tolist() * 2)
duplicated_sum_squares = sum_of_squares(california_duplicated)
print(f"복제된 캘리포니아주의 온도의 제곱합은 {duplicated_sum_squares}입니다")

value1 = california_sum_squares / california.size
value2 = duplicated_sum_squares / california_duplicated.size
assert value1 == value2

def variance(data):
    mean = np.mean(data)
    return np.mean([squared_distance(value, mean) for value in data])

assert variance(california) == california_sum_squares / california.size

assert variance(california) == variance(california_duplicated)

california_variance = variance(california)
kentucky_variance = variance(kentucky)
print(f"캘리포니아주의 온도에 대한 분산은 {california_variance}입니다")
print(f"켄터키주 온도에 대한 분산은 {kentucky_variance}입니다")

assert california_variance == california.var()
assert california_variance == np.var(california)

def weighted_variance(data, weights):
    mean = np.average(data, weights=weights)
    squared_distances = [squared_distance(value, mean) for value in data]
    return np.average(squared_distances, weights=weights)

assert weighted_variance([75, 77], [9, 1]) == np.var(9*[75] + [77])
```

### 5.3.1 확률 분포의 분산 구하기

이항 분포의 분산은 시행 횟수의 1/4

표준 편차(standard deviation)

```python
interval_all_counts = range(21)
probabilities = stats.binom.pmf(interval_all_counts, 20, prob_head)
variance_binomial = weighted_variance(interval_all_counts, probabilities)
print(f"이항 분포의 분산은 동전 앞면이 {variance_binomial:.2f}번 등장했을 때입니다.")

assert stats.binom.var(20, prob_head) == 5.0
assert stats.binom.var(20, prob_head) == stats.binom.mean(20, prob_head) / 2

variances = [stats.binom.var(num_flips, prob_head) for num_flips in range(500)]
plt.plot(range(500), variances)
plt.xlabel('동전 뒤집기')
plt.ylabel('분산')
plt.show()

assert stats.binom.var(1, 0.5) == 0.25
assert stats.binom.var(1000, 0.5) == 250

data = [1, 2, 3]
standard_deviation = np.std(data)
assert standard_deviation ** 2 == np.var(data)
```

# 6장 사이파이와 중심 극한 정리로 예측하기

## 6.1 사이파이로 정규 분포 다루기

정규 분포(normal distribution)

샘플링된 평균에 따른 상대적 확률을 히스토그램으로 그리면

-> x 값은 분포의 평균과 같고, 표준 편차는 y 값의 역에 (2파이)^(1/2)를 곱한 것과 같음

확률 질량 함수(probability mass function)

-> 최대 지점의 축 = 평균, 높이는 표준 편차에 반비례

```python
# 랜덤 시드 설정
np.random.seed(0)

# 샘플 크기 설정
sample_size = 10000

# 이항 분포 샘플 생성
sample = np.array([np.random.binomial(1, 0.5) for _ in range(sample_size)])

# 동전 앞면의 총 합
head_count = sample.sum()

# 동전 앞면이 나온 빈도 계산
head_count_frequency = head_count / sample_size

# 동전 앞면의 빈도는 평균과 같다는 것을 확인
assert head_count_frequency == sample.mean()  # head_count의 빈도는 표본 평균과 동일

# 샘플 크기 설정
np.random.seed(0)

# 샘플 10만 번에 대한 이항 분포 샘플 생성
frequencies = np.random.binomial(sample_size, 0.5, 100000) / sample_size

# 샘플 평균 계산
sample_means = frequencies

# 히스토그램 생성
likelihoods, bin_edges, _ = plt.hist(sample_means, bins='auto', edgecolor='black', density=True)

# x축, y축 라벨 설정
plt.xlabel('구간 표본 평균')
plt.ylabel('상대적 확률')

# 그래프 표시
plt.show()

# 평균 계산
mean_normal = np.average(bin_edges[:-1], weights=likelihoods)

# 분산 계산
var_normal = weighted_variance(bin_edges[:-1], likelihoods)

# 표준 편차 계산
std_normal = var_normal ** 0.5

# 출력
print(f"평균은 약 {mean_normal:.2f}입니다")
print(f"표준 편차는 약 {std_normal:.3f}입니다")

import math

# 가장 높은 지점 계산
peak_x_value = bin_edges[likelihoods.argmax()]

# 출력
print(f"평균은 약 {peak_x_value:.2f}입니다")
peak_value = likelihoods.max()
std_from_peak = (peak_x_value*(2*math.pi)**0.5) ** -1

# 표준 편차 출력
print(f"표준 편차는 약 {std_from_peak:.3f}입니다")

fitted_mean, fitted_std = stats.norm.fit(sample_means)

# 평균 출력
print(f"평균은 약 {fitted_mean:.2f}입니다")

# 표준 편차 출력
print(f"표준 편차는 약 {fitted_std:.3f}입니다")

# stats.norm.fit을 사용하여 데이터 적합
normal_likelihoods = stats.norm.pdf(bin_edges, fitted_mean, fitted_std)

# 히스토그램 그리기
plt.plot(bin_edges, normal_likelihoods, color='k', linestyle='--', label='정규 확률 곡선')
plt.hist(sample_means, bins='auto', alpha=0.2, color='r', density=True)

# 레이블 설정
plt.legend()
plt.xlabel('표본 평균')
plt.ylabel('상대적 확률')

# 그래프 표시
plt.show()

# 정규 분포 곡선 조정
adjusted_likelihoods = stats.norm.pdf(bin_edges, fitted_mean + 0.01, fitted_std / 2)

# 그래프 그리기
plt.plot(bin_edges, adjusted_likelihoods, color='k', linestyle='--')
plt.plot(bin_edges, adjusted_likelihoods, color='r', alpha=0.2)

# 레이블 설정
plt.xlabel('표본 평균')
plt.ylabel('상대적 확률')

# 그래프 표시
plt.show()
```

### 6.1.1 샘플링된 정규 분포 곡선 두 개 비교하기

같은 사건에 대해 표본 수를 4배 가량 늘리면

-> 두 분포 모두 중심은 평균과 같지만

'표본 수를 늘린 분포'가 중심에 퍼진 정도가 촘촘하며 최대 지점의 크기가 2배 가량 높음

신뢰 구간(confidence interval)

베르누이 분포(Bernoulli distribution)

포아송 분포(Poisson distribution)

감마 분포(gamma distribution)

로그 정규 분포(lognormal distribution)
```python
# 랜덤 시드 설정
np.random.seed(0)

# 새로운 샘플 크기 설정
new_sample_size = 40000

# 새로운 이항 분포 샘플 생성
new_head_counts = np.random.binomial(new_sample_size, 0.5, 100000)

# 새로운 샘플 평균과 표준편차 계산
new_mean, new_std = stats.norm.fit(new_head_counts / new_sample_size)

# 정규 분포의 가능도 계산
new_likelihoods = stats.norm.pdf(bin_edges, new_mean, new_std)

# 그래프 그리기
plt.plot(bin_edges, normal_likelihoods, color="k", linestyle='--', label='A: 표본 수 10K')
plt.plot(bin_edges, new_likelihoods, color="b", label="B: 표본 수 40K")

# 레이블 추가
plt.legend()
plt.xlabel('표본 평균')
plt.ylabel('상대적 확률')

# 그래프 표시
plt.show()

# 새로운 평균과 표준편차 설정
mean, std = new_mean, new_std

# 신뢰 구간 계산
start, end = stats.norm.interval(0.95, mean, std)

# 결과 출력
print(f"이항 분포로 샘플링된 샘 평균은 {start:.3f}와 {end:.3f} 사이에 있습니다.")

# 벤누이 분포 평균 확인
assert stats.binom.mean(1, 0.5) == 0.5

# 분산 비율 계산

new_std = stats.norm.fit(head_counts / reduced_sample_size)
variance_ratio = (std**2) / (fitted_std**2)

print(f"분산의 비율은 약 {variance_ratio:.1f}입니다")

estimated_variance = (fitted_std**2) * 10000

# 결과 출력
```

## 6.2 무작위 샘플링으로 모집단의 평균 및 분산 결정하기

모집단 평균, 모집단 분산

표본 평균, 표본 분산

중심 극한 정리(central limit theorem)에 의해 표본 평균의 히스토그램은 정규 분포와 유사

또한 정규 분포 평균은 모집단 평균에 가까움

```python
# 랜덤 시드를 설정하여 결과의 재현성을 확보합니다.
np.random.seed(0)

# 1부터 85까지의 나이로 구성된 5만 명의 데이터 샘플을 생성합니다.
population_ages = np.random.randint(1, 85, size=50000)

# 모집단 나이의 평균과 분산을 계산합니다.
population_mean = population_ages.mean()
population_variance = population_ages.var()

# 랜덤 시드를 설정하여 결과의 재현성을 확보합니다.
np.random.seed(0)

# 샘플 크기를 설정하고 모집단에서 샘플을 무작위로 선택합니다.
sample_size = 10
sample = np.random.choice(population_ages, size=sample_size)

# 선택된 샘플의 평균을 계산합니다.
sample_mean = sample.mean()

# 표본 평균과 모집단 평균의 차이를 백분율로 계산하는 함수 정의
percent_diff = lambda v1, v2: 100 * abs(v1 - v2) / v2

# 샘플 평균과 모집단 평균의 차이를 계산합니다.
percent_diff_means = percent_diff(sample_mean, population_mean)

# 결과 출력
print(f"평균 {percent_diff_means:.2f}% 차이가 있습니다")

# 랜덤 시드를 설정하여 결과의 재현성을 확보합니다.
np.random.seed(0)

# 표본 평균을 계산하기 위해 모집단에서 샘플을 선택합니다.
sample_means = [np.random.choice(population_ages, size=sample_size).mean() for _ in range(100)]

# 표본 평균들의 히스토그램을 그립니다.
likelihoods, bin_edges, _ = plt.hist(sample_means, bins='auto', alpha=0.2, color='r', density=True)

# 표본 평균들의 정규 분포를 추정하기 위해 정규 분포의 평균과 표준편차를 계산합니다.
mean, std = stats.norm.fit(sample_means)

# 계산된 평균과 표준편차를 이용해 정규 분포를 그립니다.
normal_likelihoods = stats.norm.pdf(bin_edges, mean, std)

# 히스토그램 위에 정규 분포 곡선을 그립니다.

# 실제 인구 평균을 출력합니다.

# 표본 평균과 모집단 평균을 비교하여 차이를 계산합니다.
```

## 6.3 평균과 분산을 이용하여 예측하기

평균 표준 오차(Standard Error of the Mean, SEM)

```python
population_mean = 84  # 모집단 평균

# 평균 및 SEM을 사용하여 정규 곡선 그리기
mean = population_mean  # 평균
population_std = population_variance ** 0.5  # 모집단 표준편차
sem = population_std / (20 ** 0.5)  # SEM = population_std를 표본 크기의 제곱근으로 나눈 값

# grade_range는 0부터 100까지의 범위

plt.plot(grade_range, normal_likelihoods)
```

# 9.2 평균의 차이로 p-값 계산하기

```python
# 'Blue' 컬럼을 삭제한 뒤 테이블을 전치(transpose)하여 df_not_blue에 저장
df_not_blue = df.T.drop(columns='Blue')

# df_not_blue의 상위 2개 행을 출력
print(df_not_blue.head(2))

# 두 데이터 배열 간의 순열 검정을 수행하는 함수 정의
def permutation_test(data_array_a, data_array_b):
    # 두 배열의 평균값 계산
    data_mean_a = data_array_a.mean() 
    data_mean_b = data_array_b.mean() 
    
    # 두 배열의 평균값 차이 계산
    extreme_mean_diff = abs(data_mean_a - data_mean_b)
    
    # 두 데이터 배열을 하나로 합쳐서 샘플링을 위한 데이터 배열 생성
    total_data = np.hstack([data_array_a, data_array_b]) 
    
    # 극단적인 평균 차이 값을 카운트할 변수 초기화
    number_extreme_values = 0.0 
    
    # 30000번의 샘플링 수행
    for _ in range(30000):
        np.random.shuffle(total_data)  # 데이터를 섞음
        sample_a = total_data[:data_array_a.size]  # 첫 번째 샘플
        sample_b = total_data[data_array_a.size:]  # 두 번째 샘플
        
        # 샘플 평균의 차이가 극단적인 값보다 크면 카운트 증가
        if abs(sample_a.mean() - sample_b.mean()) >= extreme_mean_diff:
            number_extreme_values += 1 

    # p-value 계산: 극단적인 값의 비율
    p_value = number_extreme_values / 30000 
    return p_value

# 순열 검정 수행 및 p-value를 색상별로 저장하는 코드
np.random.seed(0)  # 랜덤 시드 설정
blue_clicks = df.T.Blue.values  # 'Blue' 색상의 클릭 수 가져오기
color_to_p_value = {}  # 색상과 그에 대응하는 p-value를 저장할 딕셔너리

# 각 색상에 대해 순열 검정 실행
for color, color_clicks in df_not_blue.items():
    # 순열 검정 수행
    p_value = permutation_test(blue_clicks, color_clicks)
    color_to_p_value[color] = p_value  # p-value 저장

# p-value를 색상별로 정렬하여 시각화 준비
sorted_colors, sorted_p_values = zip(*sorted(color_to_p_value.items(), key=lambda x: x[1]))

# 시각화를 위한 크기 설정
plt.figure(figsize=(3, 10))

# 히트맵 시각화: p-value를 색상별로 시각화
sns.heatmap([[p_value] for p_value in sorted_p_values], cmap='YlGnBu', annot=True, xticklabels=['p-value'], yticklabels=sorted_colors)

# 그래프 출력
plt.show()

# 'Black' 색상의 평균 클릭 수 계산
mean_black = df_not_blue.Black.mean()

# 결과 출력
print(f"검은색의 평균 클릭률은 {mean_black}입니다")

# 각 색상에 대해 평균 클릭률이 파란색보다 높은 색상 필터링
remaining_colors = df[df.T.mean().values > blue_clicks.mean()].index

# 필터링된 색상 수 출력
size = remaining_colors.size
print(f"{size}개의 색상이 파란색보다 평균적으로 높은 클릭률을 가집니다")

# 필터링된 색상 목록 출력
print("구체적으로는 다음과 같습니다")
print(remaining_colors.values)

# 색상별 p-value를 계산하여 출력
for color, p_value in sorted(color_to_p_value.items(), key=lambda x: x[1]):
    if color in remaining_colors:
        mean = df_not_blue[color].mean()
        print(f"{color}는 {p_value}의 p-값과 {mean}의 평균을 가집니다")
```
