# 18장 그래프 이론 및 네트워크 분석

네트워크(=그래프): 객체와 객체의 분산된 연결 모음

네트워크 이론: 사물 간의 연결을 연구하는 학문

## 18.1 기본 그래프 이론으로 인기도별 웹 사이트 순위 지정하기

네트워크 이론 -> 공개 링크를 기반으로 웹 사이트 순위 매기는 간단한 방법 제공

노드: 서로 연결부를 형성할 수 있는 지점으로 각 웹 사이트

에지: 두 노드를 연결한 것

```python
# 노드 리스트를 정의합니다. 각 노드는 웹 사이트를 나타냅니다.
nodes = ['님파이', '사이파이']

# 노드 간의 연결을 엣지 리스트로 정의합니다.
# 1번 노드는 0번 노드로 연결됩니다.
edges = [(1, 0)]  # 여기서는 (1, 0)으로 엣지를 나타냅니다.
```

![image](https://github.com/user-attachments/assets/f93a5735-d537-4747-884d-e77e13750696)

단일 에지(=방향성 에지): nodes[1]에서 nodes[0]으로 연결된 링크

방향성 그래프: 방향성 에지를 포함한 그래프

```python
# 엣지가 존재하는지 확인하는 함수입니다.
def edge_exists(i, j):
    # (i, j)가 엣지 리스트에 있는지 확인합니다.
    return (i, j) in edges

# 엣지가 (1, 0)일 때 존재하는지 확인합니다.
assert edge_exists(1, 0)

# 엣지가 (0, 1)일 때 존재하지 않는지 확인합니다.
assert not edge_exists(0, 1)

import numpy as np

# 인접 행렬을 0으로 초기화합니다. 노드의 개수에 맞는 크기입니다.
adjacency_matrix = np.zeros((len(nodes), len(nodes)))

# edges의 정보를 바탕으로 인접 행렬을 업데이트합니다.
for i, j in edges:
    adjacency_matrix[i][j] = 1  # (i, j) 위치에 엣지가 있으면 1로 표시합니다.

# 엣지 (1, 0)가 존재하는지 확인합니다.
assert adjacency_matrix[1][0]

# 엣지 (0, 1)가 존재하지 않는지 확인합니다.
assert not adjacency_matrix[0][1]

# 최종적으로 생성된 인접 행렬을 출력합니다.
print(adjacency_matrix)
```

자기 루프(=버클): 자기 참조 에지

```python
# 인접 행렬에 (0, 1) 위치에 엣지를 추가합니다.
adjacency_matrix[0][1] = 1

# 인접 행렬을 출력합니다.
print(adjacency_matrix)
```

### 18.1.1 NetworkX로 웹 네트워크 분석하기

```python
# NetworkX 라이브러리 불러오기
import networkx as nx

# nx.DiGraph() 함수를 사용하여 방향성 그래프를 생성합니다.
G = nx.DiGraph()

# 그래프 G를 생성한 후, 그래프의 다양한 노드와 엣지 등을 추가하여 사용할 수 있습니다.

# 노드에 속성 추가 함수 정의
def print_node_attributes():
    # 그래프 G의 모든 노드를 순회하며 속성을 출력합니다.
    for i in G.nodes:
        print(f"노드 {i}의 속성 디셔너리는 {G.nodes[i]}와 같습니다.")

# 노드 속성 출력 함수 호출
print_node_attributes()

# 첫 번째 노드(0번 노드)에 'webpage' 속성 추가
G.nodes[0]['webpage'] = '넘파이'

# 웹페이지 속성을 추가한 후, 다시 노드 속성 출력
print('\n노드 0에 웹 페이지를 추가했습니다.')
print_node_attributes()

# 노드 1에 'webpage' 속성 추가
G.add_node(1, webpage='사이파이')

# 노드의 속성 출력
print_node_attributes()

# 노드와 해당 속성들을 data=True 인자를 사용하여 함께 출력
print(G.nodes(data=True))

# 'webpage' 속성을 가진 모든 노드를 함께 출력할 수 있습니다.

import matplotlib.pyplot as plt  # matplotlib의 pyplot을 plt라는 이름으로 임포트

# 랜덤 시드값 설정 - 그래프 노드들의 위치 결정에 영향을 미침
np.random.seed(0)

# 그래프 그리기
nx.draw(G)  # G 그래프의 노드들과 연결선을 시각적으로 출력

# 그래프 시각화
plt.show()  # plt.show()로 그래프를 출력
```

```python
# 'nx.to_numpy_array()'를 사용하여 그래프 G의 인접 행렬을 numpy 배열로 변환합니다.
adjacency_matrix = nx.to_numpy_array(G)

# 변환된 인접 행렬을 출력합니다.
print(adjacency_matrix)

# adjacency_matrix에서 각 노드의 인-디그리 계산 (행 방향으로 합산)
in_degrees = adjacency_matrix.sum(axis=0)

# 각 노드에 대해 인-디그리를 출력
for i, in_degree in enumerate(in_degrees):
    page = G.nodes[i]['webpage']  # 노드의 'webpage' 속성 값 가져오기
    print(f'{page} 페이지의 인-디그리는 {in_degree}입니다.')

# 가장 인-디그리가 높은 페이지 찾기
top_page = G.nodes[in_degrees.argmax()]['webpage']
print(f'\n{top_page} 페이지가 가장 인기 있습니다.')

# NetworkX 라이브러리를 사용하여 노드 0의 인-디그리가 2인지 확인
assert G.in_degree(0) == 2  # G의 노드 0의 인-디그리 값이 2인지 검증

# 새로운 방향성 그래프 객체 G2 생성
G2 = nx.DiGraph()  # 방향성 그래프를 생성합니다.

# 노드 추가: '님파이', '사이파이', '맷플롯립', '판다스' 노드를 그래프에 추가
G2.add_nodes_from(['님파이', '사이파이', '맷플롯립', '판다스'])  # 노드를 그래프에 추가합니다.

# 엣지 추가: ('사이파이', '님파이'), ('사이파이', '맷플롯립'), ('님파이', '판다스'), ('맷플롯립', '님파이') 엣지를 그래프에 추가
G2.add_edges_from([('사이파이', '님파이'), ('사이파이', '맷플롯립'), ('님파이', '판다스'), ('맷플롯립', '님파이')])  # 엣지를 그래프에 추가합니다.

# 노드 '님파이'의 인-디그리가 2인지 확인
assert G2.in_degree('님파이') == 2  # '님파이' 노드의 인-디그리 값이 2인지 확인합니다.
```

## 18.2 비방향성 그래프로 마을 간 이동 시간 최적화하기

물류 비즈니스에서 제품 배송 시간은 중요한 의사 결정에 영향을 미침

한 도로가 마을 0과 마을 1을 서로 연결하며 두 마을 간 이동 시간은 20분이라 가정.

```python
# 방향성 없는 그래프 G 생성
G = nx.Graph()  # 방향성 없는 그래프 객체를 생성합니다.

# 엣지 추가: 0 -> 1로 이동하는 엣지 추가
G.add_edge(0, 1)  # 노드 0에서 노드 1로 이동하는 엣지를 추가합니다.

# travel_time 속성 설정: 노드 0에서 노드 1로 이동하는 travel_time을 20으로 설정
G[0][1]['travel_time'] = 20  # 엣지 (0, 1)에서 travel_time 속성을 20으로 설정합니다.

# 그래프 엣지의 이동 시간을 확인
for i, j in [(0, 1), (1, 0)]:  # 엣지 (0, 1)과 (1, 0)을 확인합니다.
    travel_time = G[i][j]['travel_time']  # travel_time 속성 값을 가져옵니다.
    print(f"마을 {i} -> 마을 {j}로의 이동 시간은 {travel_time}분 걸립니다.")  # 이동 시간을 출력합니다.
```

![image](https://github.com/user-attachments/assets/31911572-e22a-4150-930f-271223b7f69d)

```python
# 여러 마을 사이의 경로 시각화하기
np.random.seed(0)  # 랜덤 시드를 설정하여 동일한 결과를 얻을 수 있게 합니다.
G.add_edge(1, 2, travel_time=15)  # 마을 1에서 마을 2로 이동하는 엣지 추가, travel_time 속성은 15로 설정
nx.draw(G, with_labels=True, node_color='khaki')  # 그래프를 그릴 때 마을 이름과 함께 그리며, 노드 색상은 'khaki'로 설정
plt.show()  # 그래프를 화면에 출력

# 마을 0 -> 마을 2로 이동하려면 먼저 마을 1을 통과해야 합니다. 
# travel_time과 G[1][2]['travel_time'] 값을 확인하여 경로 시간을 확인
print(f"마을 0 -> 마을 2로의 이동 시간은 {G[1][2]['travel_time']}분 걸립니다.")  # 마을 0에서 마을 2로의 travel_time 출력
```
